cmake_minimum_required(VERSION 3.22)
project(Aurora VERSION 0.1.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE" CACHE PATH "JUCE 7.0.12 source")
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/zstd/build/cmake zstd)
add_subdirectory("${JUCE_ROOT}" JUCE)
juce_add_plugin(Aurora COMPANY_NAME "Independent Audio" BUNDLE_ID "audio.independent.aurora" PLUGIN_MANUFACTURER_CODE InAu PLUGIN_CODE Au01 FORMATS VST3 Standalone PRODUCT_NAME "Aurora" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE VST3_CATEGORIES Instrument Synth COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(Aurora)
target_sources(Aurora PRIVATE Source/Plugin.cpp)
target_compile_definitions(Aurora PUBLIC JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Aurora PRIVATE "$<LINK_ONLY:libzstd_static>" juce::juce_audio_utils juce::juce_dsp juce::juce_opengl PUBLIC juce::juce_recommended_config_flags)
juce_add_plugin(AuroraFX COMPANY_NAME "Independent Audio" BUNDLE_ID "audio.independent.aurorafx" PLUGIN_MANUFACTURER_CODE InAu PLUGIN_CODE Af01 FORMATS VST3 PRODUCT_NAME "Aurora FX" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE VST3_CATEGORIES Fx COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(AuroraFX)
target_sources(AuroraFX PRIVATE Source/Plugin.cpp)
target_compile_definitions(AuroraFX PUBLIC AURORA_FX=1 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(AuroraFX PRIVATE "$<LINK_ONLY:libzstd_static>" juce::juce_audio_utils juce::juce_dsp juce::juce_opengl PUBLIC juce::juce_recommended_config_flags)
juce_add_console_app(AuroraTests PRODUCT_NAME "Aurora DSP tests")
juce_generate_juce_header(AuroraTests)
target_sources(AuroraTests PRIVATE Source/Tests.cpp Source/Plugin.cpp)
target_compile_definitions(AuroraTests PRIVATE JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1 JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(AuroraTests PRIVATE "$<LINK_ONLY:libzstd_static>" juce::juce_audio_utils juce::juce_dsp juce::juce_opengl)
# JUCE 7's resource rule does not depend on metadata. Refresh Windows file
# versions when project metadata changes in an existing build directory.
if(WIN32)
    foreach(target Aurora AuroraFX AuroraTests)
        get_target_property(code_dir ${target} JUCE_GENERATED_SOURCES_DIRECTORY)
        get_target_property(info_file ${target} JUCE_INFO_FILE)
        add_custom_command(OUTPUT "${code_dir}/${target}_resources.rc" APPEND DEPENDS "${info_file}")
    endforeach()
endif()
if(MINGW)
    foreach(target Aurora_VST3 AuroraFX_VST3 Aurora_Standalone AuroraTests)
        target_link_options(${target} PRIVATE -static)
    endforeach()
endif()
